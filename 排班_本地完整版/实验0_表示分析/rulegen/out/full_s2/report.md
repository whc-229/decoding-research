## r52（得分 0.8210）

The hybrid keeps Rule 1’s exact single-type behavior (where it scored best) and its feasibility-aware mixed-duty primary choice, while using Rule 2’s maintenance-aware late-horizon ordering and load-balanced alternative ordering for multi-type instances. In non-late multi-type shifts it preserves Rule 1’s more balanced primary robot but exposes Rule 2’s ordering as deviation alternatives; in the endgame it prioritizes maintenance-aware reuse and includes mixed-duty horizon feasibility checks.

def rule(ctx):

    n = len(types)

    def params(k):
        ti = types[k] if 0 <= k < n else None
        if ti is None:
            return 10.0, 0.0, 0.0, 0.0, 0.0, 10.0, 1.0
        wt = ti.get("work_time", 10)
        if wt is None:
            wt = 10
        wt = float(wt)
        out = ti.get("out_travel_time", 0)
        out = float(out) if out is not None else 0.0
        ret = ti.get("return_travel_time", 0)

        ch = ti.get("charge_time", 0)
        ch = float(ch) if ch is not None else 0.0

        mt = float(mt) if mt is not None else 0.0
        single = ti.get("single_work_time", wt)
        if single is None or single <= 0:
            single = wt if wt > 0 else 1.0
        else:
            single = float(single)
        cont = ti.get("continue_max_work", 0)

            cont = 0.0
        cont = float(cont)

                fit_val = rrem

    avail = list(ctx.available)
    if not avail:
        return new_candidates(ctx.remaining)

    if n_active == 1:
        ordered = rule1_order(avail)
        actions = [("reuse", r) for r in ordered]
        actions.extend(new_candidates(ctx.remaining))
        return actions

    late = ctx.remaining <= max_maint + 2.0 * max_cyc + 1e-9
    if late:

        primary = ordered[0]
        rest = ordered[1:]
    else:
        r1 = rule1_order(avail)
        primary = r1[0]
        r2 = rule2_nonlate_order(avail)
        rest = [r for r in r2 if r.idx != primary.idx]

    actions = [("reuse", primary)]
    for r in rest:
        actions.append(("reuse", r))
    actions.extend(new_candidates(ctx.remaining))
    return actions

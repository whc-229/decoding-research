## e52（得分 0.8162）

Maintenance-first fairness band: within one duty of the current minimum count, prefer robots closest to maintenance while still fitting the remaining horizon, and order new types by horizon fit, gene preference, and productive-cycle efficiency.

```python
def rule(ctx):
    available = ctx.available
    if available:
        min_count = min(r.count for r in available)
    else:
        min_count = 0

    def reuse_key(r):
        p = ctx.types[r.type]
        if p is not None:
            wt = p["work_time"]
        else:
            wt = 10.0
        over = wt - ctx.remaining
        if over < 0:
            over = 0.0
        band = r.count - min_count
        if band < 0:
            band = 0
        if band <= 1:
            band_pen = 0
        else:
            band_pen = band
        return (band_pen, over, r.duties_left, r.hours, -r.budget, r.idx)

    reuses = sorted(available, key=reuse_key)

    remaining = ctx.remaining
    if remaining < 0:
        remaining = 0.0

    def type_key(k):
        p = ctx.types[k]
        if p is None or ctx.left[k] <= 0:
            return (1, 0, 0, 0, 0, 0, 0, 0, k)
        wt = p["work_time"]
        if wt <= 0:
            wt = 1.0
        over = wt - remaining
        if over < 0:
            over = 0.0
        if remaining > 0:
            waste = (remaining % wt) / wt
        else:
            waste = 0.0
        single = p["single_work_time"]
        cmw = p["continue_max_work"]
        mt = p["maintenance_time"]
        ct = p["charge_time"]
        out = p["out_travel_time"]
        ret = p["return_travel_time"]
        if single > 0:
            dpm = cmw / single
        else:
            dpm = 1.0
        if dpm < 1.0:
            dpm = 1.0
        denom = out + wt + ret + ct + mt / dpm
        if denom <= 0:
            denom = 1.0
        eff = wt / denom
        gene_pen = 0 if k == ctx.gene_type else 1
        return (1 if over > 0 else 0, over, gene_pen, waste, -eff, mt / dpm, -dpm, -wt, k)

    new_order = []
    for k in sorted(range(len(ctx.types)), key=type_key):
        p = ctx.types[k]
        if p is not None and ctx.left[k] > 0:
            new_order.append(k)

    actions = [("reuse", r) for r in reuses]
    for k in new_order:
        actions.append(("new", k))

    if not new_order:
        gt = ctx.gene_type
        if 0 <= gt < len(ctx.types):
            actions.append(("new", gt))

    return actions
```

## e39（得分 0.7804）

Prioritize available robots by a weighted balance score comparing their duty count and hours to the fleet average, penalizing imminent maintenance, and order new types by gene preference, horizon fit, and amortized maintenance overhead.

```python
def rule(ctx):
    n = len(ctx.robots)
    if n > 0:
        total_count = sum(r.count for r in ctx.robots)
        total_hours = sum(r.hours for r in ctx.robots)
        avg_count = total_count / n
        avg_hours = total_hours / n
    else:
        avg_count = 0.0
        avg_hours = 0.0

    def reuse_key(r):
        p = ctx.types[r.type]
        w = p["work_time"] if p is not None else 10.0
        if w <= 0:
            w = 10.0
        count_gap = r.count - avg_count
        hours_gap = r.hours - avg_hours
        maint_penalty = 0.0
        if r.duties_left <= 1:
            maint_penalty += 5.0
        if r.first_maint is not None:
            maint_penalty += 0.1
        cost = count_gap + (hours_gap / w) - 0.2 * r.duties_left + maint_penalty
        return (cost, r.ready, r.idx)

    reuses = sorted(ctx.available, key=reuse_key)

    remaining = ctx.remaining
    if remaining < 0:
        remaining = 0.0

    new_candidates = []
    for k in range(len(ctx.types)):
        p = ctx.types[k]
        if p is None or ctx.left[k] <= 0:
            continue
        wt = p["work_time"]
        cmw = p["continue_max_work"]
        mt = p["maintenance_time"]
        ct = p["charge_time"]
        single = p["single_work_time"]
        if single > 0:
            duties_per_cycle = cmw / single
        else:
            duties_per_cycle = 1.0
        if duties_per_cycle < 1.0:
            duties_per_cycle = 1.0
        total_time = p["out_travel_time"] + wt + p["return_travel_time"] + ct + mt / duties_per_cycle
        if wt > remaining:
            fit_penalty = (wt - remaining) * 10.0
        else:
            fit_penalty = 0.0
        if wt > 0:
            waste = (remaining % wt) / wt
        else:
            waste = 0.0
        gene_penalty = 0.0 if k == ctx.gene_type else 100.0
        cost = gene_penalty + fit_penalty + total_time + waste * 5.0
        new_candidates.append((cost, k))

    new_candidates.sort()
    new_types = [k for _, k in new_candidates]

    if not new_types and ctx.gene_type is not None and 0 <= ctx.gene_type < len(ctx.types):
        new_types.append(ctx.gene_type)

    actions = [("reuse", r) for r in reuses]
    for k in new_types:
        actions.append(("new", k))
    return actions
```

def rule(ctx):
    def robot_key(r):
        p = ctx.types[r.type]
        w = p["work_time"]
        pc = r.count + 1
        ph = r.hours + w
        risk = 1 if (r.first_maint is None and r.duties_left <= 1) else 0
        return (pc, risk, ph, -r.duties_left, -r.budget, r.idx)

    reuses = sorted(ctx.available, key=robot_key)

    new_types = []
    if ctx.gene_type is not None and 0 <= ctx.gene_type < len(ctx.types):
        if ctx.types[ctx.gene_type] is not None and ctx.left[ctx.gene_type] > 0:
            new_types.append(ctx.gene_type)

    others = []
    for k in range(len(ctx.types)):
        p = ctx.types[k]
        if p is None or ctx.left[k] <= 0:
            continue
        if k == ctx.gene_type:
            continue
        w = p["work_time"]
        rem = ctx.remaining
        if w <= rem:
            fit = 0
            overshoot = -w
        else:
            fit = 1
            overshoot = w
        others.append((fit, overshoot, p["maintenance_time"], p["charge_time"], k))
    others.sort()
    for item in others:
        new_types.append(item[4])

    if not new_types and ctx.gene_type is not None and 0 <= ctx.gene_type < len(ctx.types):
        new_types.append(ctx.gene_type)

    return [("reuse", r) for r in reuses] + [("new", k) for k in new_types]
